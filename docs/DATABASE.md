# Database Schema & Data Models

Sumora uses **PostgreSQL** with **Prisma ORM**.

## Entity Relationship Overview

```
User (Account)
  ├── UserIntegration (YouTube OAuth Tokens, refresh token, expiry)
  ├── SubscriptionSource (Channels subscribed to, e.g. YouTube Channel ID)
  │     └── Video (Uploaded videos metadata)
  │           └── Transcript (Raw captions & normalized transcript)
  │                 └── Summary (AI generated key takeaways, highlights)
  ├── Notification (FCM/APNs push history, read status)
  └── UserPreference (AI depth, digest frequency, topic filters)
```

## Prisma Schema Draft

```prisma
datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

generator client {
  provider = "prisma-client-js"
}

model User {
  id           String            @id @default(uuid())
  email        String            @unique
  name         String?
  createdAt    DateTime          @default(now())
  updatedAt    DateTime          @updatedAt
  integrations UserIntegration[]
  preferences  UserPreference?
  notifications Notification[]
}

model UserIntegration {
  id           String    @id @default(uuid())
  userId       String
  user         User      @relation(fields: [userId], references: [id], onDelete: Cascade)
  provider     String    // 'youtube'
  accessToken  String
  refreshToken String?
  tokenExpiry  DateTime?
  scope        String?
  connectedAt  DateTime  @default(now())
  updatedAt    DateTime  @updatedAt

  @@unique([userId, provider])
}

model SubscriptionSource {
  id           String    @id @default(uuid())
  platform     String    // 'youtube'
  externalId   String    // channelId
  name         String
  thumbnailUrl String?
  createdAt    DateTime  @default(now())
  updatedAt    DateTime  @updatedAt
  videos       Video[]

  @@unique([platform, externalId])
}

model Video {
  id             String             @id @default(uuid())
  sourceId       String
  source         SubscriptionSource @relation(fields: [sourceId], references: [id], onDelete: Cascade)
  platformVideoId String            // YouTube Video ID
  title          String
  publishedAt    DateTime
  duration       Int?               // in seconds
  thumbnailUrl   String?
  transcript     Transcript?
  summary        Summary?
  createdAt      DateTime           @default(now())

  @@unique([sourceId, platformVideoId])
}

model Transcript {
  id          String   @id @default(uuid())
  videoId     String   @unique
  video       Video    @relation(fields: [videoId], references: [id], onDelete: Cascade)
  rawText     String   @db.Text
  normalized  String   @db.Text
  language    String   @default("en")
  isAutoGenerated Boolean @default(false)
  createdAt   DateTime @default(now())
}

model Summary {
  id          String   @id @default(uuid())
  videoId     String   @unique
  video       Video    @relation(fields: [videoId], references: [id], onDelete: Cascade)
  model       String   // e.g. "gemini-1.5-flash", "gpt-4o"
  headline    String
  keyPoints   Json     // Array of points
  actionItems Json?    // Array of actionable takeaways
  readingTime Int      // estimated in minutes
  createdAt   DateTime @default(now())
}

model Notification {
  id        String   @id @default(uuid())
  userId    String
  user      User     @relation(fields: [userId], references: [id], onDelete: Cascade)
  title     String
  body      String
  data      Json?
  isRead    Boolean  @default(false)
  createdAt DateTime @default(now())
}

model UserPreference {
  id               String   @id @default(uuid())
  userId           String   @unique
  user             User     @relation(fields: [userId], references: [id], onDelete: Cascade)
  summaryLength    String   @default("medium") // short, medium, detailed
  notificationEnabled Boolean @default(true)
  createdAt        DateTime @default(now())
  updatedAt        DateTime @updatedAt
}
```
